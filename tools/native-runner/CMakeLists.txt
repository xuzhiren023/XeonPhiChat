cmake_minimum_required(VERSION 3.16)
project(knc_qualification C CXX ASM)
set(CMAKE_CXX_STANDARD 14)
add_subdirectory(${KNC_SOURCE} source)
add_executable(phi-modern main.cpp)
target_include_directories(phi-modern PRIVATE ${KNC_SOURCE}/ggml/src/ggml-cpu/arch/knc)
target_compile_definitions(phi-modern PRIVATE PHI_PROFILE="${GGML_KNC_PROFILE}" PHI_KNC_KERNELS PHI_LATEST_REFERENCE)
target_link_libraries(phi-modern PRIVATE llama)
target_link_options(phi-modern PRIVATE -fopenmp)
set_target_properties(phi-modern PROPERTIES BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "$ORIGIN/../lib")
